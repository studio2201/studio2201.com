02 · GITHUB ACTIONS & CLI

GitHub Actions & CI enforcement.

Audit repository debt catalogs on pull requests and scheduled runs using the standalone Boneyard action.

# .github/workflows/boneyard.yml
name: boneyard
on:
  push: { branches: [ master, main ] }
  pull_request: { branches: [ master, main ] }
  workflow_dispatch:
concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: ${{ github.event_name == 'pull_request' }}
permissions: { contents: read }
jobs:
  boneyard:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v6
        with: { fetch-depth: 0 }
      - uses: studio2201/boneyard@v1
      - uses: actions/upload-artifact@v6
        if: always()
        with: { name: boneyard-report, path: boneyard_summary.md, if-no-files-found: ignore }
# Workstation CLI installation:
cargo install --git https://github.com/studio2201/boneyard
boneyard enrich --input repos.json --output scored.json
