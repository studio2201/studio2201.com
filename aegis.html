02 · GITHUB ACTIONS & CLI

GitHub Actions & CI enforcement.

Prevent classical cryptographic regressions and enforce NIST FIPS 203/204 compliance on every pull request.

# .github/workflows/aegis.yml
name: aegis
on:
  push: { branches: [ master, main ] }
  pull_request: { branches: [ master, main ] }
  workflow_dispatch:
concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: ${{ github.event_name == 'pull_request' }}
permissions: { contents: read }
jobs:
  aegis:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v6
        with: { fetch-depth: 0 }
      - uses: studio2201/aegis@v1
      - uses: actions/upload-artifact@v6
        if: always()
        with: { name: aegis-report, path: aegis_summary.md, if-no-files-found: ignore }
# Workstation CLI installation:
cargo install --git https://github.com/studio2201/aegis
aegis scan --src ./src && aegis check --policy strict-pqc
